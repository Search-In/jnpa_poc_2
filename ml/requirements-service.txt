# --- HTTP layer — genuinely required -------------------------------------
# api_uc2.py and every build_router() need these. Without them the modules
# still run as CLIs but nothing is served.
fastapi>=0.115
pydantic>=2.7
uvicorn[standard]>=0.30

# --- M1 / M2 / M3 learned engines ----------------------------------------
# Fallback chain (src/uc2_models/uc2_learn.py):
#   HistGradientBoosting -> GradientBoosting -> RandomForest -> a stdlib ridge
#   solved by Gauss-Jordan on the normal equations.
# OMITTING scikit-learn: the three learned models still train and still report
# honest held-out metrics on the stdlib ridge — degraded, and badged as such in
# `engine_trace` and in `degraded`. Kept in, because the panel's headline M1
# number is the one an operator reads first and the ridge is materially worse.
scikit-learn>=1.3

# --- Legacy .xls shipping-line inventories -------------------------------
# Two of the nine EAL/IAL exports are pre-2007 .xls (EAL_NSICT, EAL_NSIGT).
# OMITTING xlrd: they are skipped, named in the provenance, and M7 reports
# `degraded: true` rather than balancing a pool missing two terminals.
# Kept in — but note this checkout excludes the corpus entirely, so M7 is
# degraded here regardless (see data/corpus/README.md).
xlrd>=2.0
